/** * Availability Survey — backend (multi-poll) * ---------------------------------------------- * One deployment backs any number of polls, all listed and managed from * inside the tool itself (create / rename / delete — no per-poll setup * needed here). * * Surveys sheet — one row per poll: survey_id, config_json, updated_at * Responses: — one tab per poll, auto-created, one row per * (person, marked slot) — easy to open and pivot on. * * Deploy: Extensions > Apps Script > paste this in > Deploy > New * deployment > type "Web app" > Execute as: Me > Who has access: Anyone. * After editing this file, you must push a NEW deployment VERSION * (Deploy > Manage deployments > pencil icon > New version) for changes * to reach the existing Web app URL. */ var SURVEYS_SHEET = 'Surveys'; var RESPONSES_PREFIX = 'Responses: '; function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var params = (e && e.parameter) || {}; var action = params.action; var surveyId = params.survey; if (action === 'listSurveys') { return jsonOut_({ surveys: listSurveys_(ss) }); } if (action === 'getConfig') { if (!surveyId) return jsonOut_({ error: 'missing survey id' }); return jsonOut_({ config: readConfig_(ss, surveyId) }); } if (action === 'listResponses') { if (!surveyId) return jsonOut_({ error: 'missing survey id' }); return jsonOut_({ responses: readResponses_(ss, surveyId) }); } return jsonOut_({ error: 'unknown action', hint: 'Use ?action=listSurveys, ?action=getConfig&survey=ID, or ?action=listResponses&survey=ID.' }); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); if (!e || !e.postData) { return jsonOut_({ ok: false, error: 'no post data — this endpoint expects a POST request with a JSON body, not a manual Run.' }); } var body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonOut_({ ok: false, error: 'bad request' }); } var action = body.action; var payload = body.payload || {}; var surveyId = payload.surveyId; if (!surveyId) { return jsonOut_({ ok: false, error: 'missing survey id' }); } if (action === 'saveConfig') { writeConfig_(ss, surveyId, payload); return jsonOut_({ ok: true }); } if (action === 'saveResponse') { appendResponse_(ss, surveyId, payload); return jsonOut_({ ok: true }); } if (action === 'clearResponses') { var cfg = readConfig_(ss, surveyId); if (cfg && cfg.passphrase && cfg.passphrase !== payload.passphrase) { return jsonOut_({ ok: false, error: 'unauthorized' }); } clearResponses_(ss, surveyId); return jsonOut_({ ok: true }); } if (action === 'deleteSurvey') { var cfg2 = readConfig_(ss, surveyId); if (cfg2 && cfg2.passphrase && cfg2.passphrase !== payload.passphrase) { return jsonOut_({ ok: false, error: 'unauthorized' }); } deleteSurvey_(ss, surveyId); return jsonOut_({ ok: true }); } return jsonOut_({ ok: false, error: 'unknown action' }); } // ---------- helpers ---------- function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function getSheet_(ss, name, headers) { var sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.appendRow(headers); } return sh; } // Google Sheets tab names can't contain []*?/\: and are capped at 100 chars. function responsesSheetName_(surveyId) { var clean = String(surveyId).replace(/[\[\]\*\?\/\\:]/g, '-'); var name = RESPONSES_PREFIX + clean; return name.substring(0, 100); } function listSurveys_(ss) { var sh = getSheet_(ss, SURVEYS_SHEET, ['survey_id', 'config_json', 'updated_at']); var values = sh.getDataRange().getValues(); var out = []; for (var i = 1; i < values.length; i++) { var surveyId = values[i][0]; if (!surveyId) continue; var cfg; try { cfg = JSON.parse(values[i][1]); } catch (err) { continue; } out.push({ surveyId: surveyId, title: cfg.title, description: cfg.description, tags: cfg.tags || [], dates: cfg.dates, startMinutes: cfg.startMinutes, endMinutes: cfg.endMinutes, tz: cfg.tz, hasPassphrase: !!cfg.passphrase, updatedAt: values[i][2], responseCount: countRespondents_(ss, surveyId), }); } out.sort(function (a, b) { return a.updatedAt < b.updatedAt ? 1 : -1; }); return out; } function countRespondents_(ss, surveyId) { var sh = ss.getSheetByName(responsesSheetName_(surveyId)); if (!sh) return 0; var lastRow = sh.getLastRow(); if (lastRow < 2) return 0; var ids = sh.getRange(2, 1, lastRow - 1, 1).getValues(); var seen = {}; ids.forEach(function (r) { if (r[0]) seen[r[0]] = true; }); return Object.keys(seen).length; } function readConfig_(ss, surveyId) { var sh = getSheet_(ss, SURVEYS_SHEET, ['survey_id', 'config_json', 'updated_at']); var values = sh.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][0] === surveyId) { try { return JSON.parse(values[i][1]); } catch (err) { return null; } } } return null; } function writeConfig_(ss, surveyId, config) { var sh = getSheet_(ss, SURVEYS_SHEET, ['survey_id', 'config_json', 'updated_at']); var values = sh.getDataRange().getValues(); var now = new Date().toISOString(); for (var i = 1; i < values.length; i++) { if (values[i][0] === surveyId) { sh.getRange(i + 1, 2, 1, 2).setValues([[JSON.stringify(config), now]]); return; } } sh.appendRow([surveyId, JSON.stringify(config), now]); } function readResponses_(ss, surveyId) { var sh = ss.getSheetByName(responsesSheetName_(surveyId)); if (!sh) return []; var values = sh.getDataRange().getValues(); var rows = values.slice(1).filter(function (r) { return r[0]; }); var map = {}; rows.forEach(function (r) { var id = r[0]; if (!map[id]) map[id] = { id: id, name: r[1], tz: r[2], submittedAt: r[3], slots: [] }; map[id].slots.push(r[4]); }); return Object.keys(map).map(function (k) { return map[k]; }); } function appendResponse_(ss, surveyId, record) { var sh = getSheet_(ss, responsesSheetName_(surveyId), ['id', 'name', 'tz', 'submitted_at', 'slot_utc_iso']); var slots = record.slots || []; if (!slots.length) return; var rows = slots.map(function (slot) { return [record.id, record.name, record.tz, record.submittedAt, slot]; }); sh.getRange(sh.getLastRow() + 1, 1, rows.length, 5).setValues(rows); } function clearResponses_(ss, surveyId) { var sh = ss.getSheetByName(responsesSheetName_(surveyId)); if (!sh) return; var lastRow = sh.getLastRow(); if (lastRow > 1) sh.deleteRows(2, lastRow - 1); } function deleteSurvey_(ss, surveyId) { var sh = getSheet_(ss, SURVEYS_SHEET, ['survey_id', 'config_json', 'updated_at']); var values = sh.getDataRange().getValues(); for (var i = values.length - 1; i >= 1; i--) { if (values[i][0] === surveyId) { sh.deleteRow(i + 1); break; } } var respSheet = ss.getSheetByName(responsesSheetName_(surveyId)); if (respSheet) ss.deleteSheet(respSheet); }
